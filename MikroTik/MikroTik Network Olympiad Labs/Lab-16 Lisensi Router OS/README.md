<h3>Level Lisensi MikroTik</h3>

<table>
  <tr>
    <th>Level</th>
    <th>Keterangan</th>
    <th>Kegunaan</th>
  </tr>

  <tr>
    <td><b>Level 0</b></td>
    <td>Trial</td>
    <td>Digunakan untuk mencoba RouterOS dalam waktu terbatas sebelum menggunakan lisensi penuh.</td>
  </tr>

  <tr>
    <td><b>Level 1</b></td>
    <td>Free/Demo</td>
    <td>Memiliki fitur dasar RouterOS dan dapat digunakan secara gratis dengan beberapa batasan.</td>
  </tr>

  <tr>
    <td><b>Level 2</b></td>
    <td>Deprecated</td>
    <td>Merupakan level lisensi lama yang sudah tidak digunakan pada RouterOS versi sekarang.</td>
  </tr>

  <tr>
    <td><b>Level 3</b></td>
    <td>WISP CPE</td>
    <td>Digunakan untuk perangkat sebagai wireless client atau station, terutama pada penggunaan CPE.</td>
  </tr>

  <tr>
    <td><b>Level 4</b></td>
    <td>WISP</td>
    <td>Dapat digunakan sebagai wireless AP maupun station dan mendukung berbagai fitur routing serta jaringan.</td>
  </tr>

  <tr>
    <td><b>Level 5</b></td>
    <td>WISP</td>
    <td>Memiliki kemampuan wireless yang lebih tinggi dibandingkan Level 4 dan dapat digunakan untuk jaringan dengan kebutuhan lebih besar.</td>
  </tr>

  <tr>
    <td><b>Level 6</b></td>
    <td>Controller</td>
    <td>Merupakan level lisensi tertinggi dengan fitur dan batas penggunaan yang paling lengkap.</td>
  </tr>
</table>
